-- --------------------------------------------------------
-- Tabla `cliente`  (con CHECK de formato y persona/empresa)
-- --------------------------------------------------------
CREATE TABLE `cliente` (
  `id` bigint(20) NOT NULL,
  `tipo_documento` enum('DNI','PASAPORTE','RUC') NOT NULL,
  `numero_documento` varchar(20) NOT NULL,
  `nombres` varchar(100) DEFAULT NULL,
  `apellidos` varchar(100) DEFAULT NULL,
  `razon_social` varchar(150) DEFAULT NULL,
  `celular` varchar(15) NOT NULL,
  `direccion` varchar(200) DEFAULT NULL,
  `fecha_creacion` datetime NOT NULL DEFAULT current_timestamp(),
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_cliente_documento` (`tipo_documento`,`numero_documento`),
  CONSTRAINT `chk_cliente_doc_formato` CHECK (
    (`tipo_documento` = 'DNI'       AND `numero_documento` REGEXP '^[0-9]{8}$')
    OR
    (`tipo_documento` = 'RUC'       AND `numero_documento` REGEXP '^[0-9]{11}$')
    OR
    (`tipo_documento` = 'PASAPORTE' AND `numero_documento` REGEXP '^[A-Za-z0-9]{6,20}$')
  ),
  CONSTRAINT `chk_cliente_persona_empresa` CHECK (
    (`tipo_documento` = 'RUC' AND `razon_social` IS NOT NULL)
    OR
    (`tipo_documento` IN ('DNI','PASAPORTE') AND `nombres` IS NOT NULL AND `apellidos` IS NOT NULL)
  )
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- --------------------------------------------------------
-- Tabla `tarifa`  (con UNIQUE para evitar duplicados activos)
-- --------------------------------------------------------
CREATE TABLE `tarifa` (
  `id` bigint(20) NOT NULL,
  `tipo_paquete_id` bigint(20) NOT NULL,
  `agencia_destino_id` bigint(20) DEFAULT NULL,
  `precio_base` decimal(10,2) NOT NULL DEFAULT 0.00,
  `precio_por_kg` decimal(10,2) NOT NULL DEFAULT 0.00,
  `activa` tinyint(1) NOT NULL DEFAULT 1,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_tarifa_tipo_destino_activa` (`tipo_paquete_id`,`agencia_destino_id`,`activa`),
  KEY `fk_tarifa_destino` (`agencia_destino_id`),
  CONSTRAINT `fk_tarifa_tipo` FOREIGN KEY (`tipo_paquete_id`) REFERENCES `tipo_paquete` (`id`),
  CONSTRAINT `fk_tarifa_destino` FOREIGN KEY (`agencia_destino_id`) REFERENCES `agencia` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- --------------------------------------------------------
-- Tabla `usuario`  (seeds al final)
-- --------------------------------------------------------
CREATE TABLE `usuario` (

-- --------------------------------------------------------
-- Tabla `encomienda`
--   * registrado_por_usuario_id NOT NULL
--   * anulado + fecha_anulacion (Opción B)
--   * CHECK de coherencia estado <-> fecha_recepcion/costo_total
-- --------------------------------------------------------
CREATE TABLE `encomienda` (
  `id` bigint(20) NOT NULL,
  `codigo_seguimiento` varchar(20) NOT NULL,
  `canal` enum('WEB','AGENCIA') NOT NULL,
  `estado` enum('PRE_REGISTRADO','EN_AGENCIA','EN_TRANSITO','LLEGO_A_DESTINO','RECOGIDO') NOT NULL DEFAULT 'PRE_REGISTRADO',
  `remitente_id` bigint(20) NOT NULL,
  `destinatario_id` bigint(20) NOT NULL,
  `agencia_origen_id` bigint(20) NOT NULL,
  `agencia_destino_id` bigint(20) NOT NULL,
  `contacto_recojo_nombre` varchar(100) DEFAULT NULL,
  `contacto_recojo_celular` varchar(15) DEFAULT NULL,
  `costo_total` decimal(10,2) DEFAULT NULL,
  `registrado_por_usuario_id` bigint(20) NOT NULL,
  `fecha_registro` datetime NOT NULL DEFAULT current_timestamp(),
  `fecha_recepcion` datetime DEFAULT NULL,
  `empleado_recepcion_id` bigint(20) DEFAULT NULL,
  `recogido_por_documento` varchar(20) DEFAULT NULL,
  `recogido_por_nombre` varchar(150) DEFAULT NULL,
  `fecha_entrega` datetime DEFAULT NULL,
  `empleado_entrega_id` bigint(20) DEFAULT NULL,
  `anulado` tinyint(1) NOT NULL DEFAULT 0,
  `fecha_anulacion` datetime DEFAULT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_encomienda_codigo` (`codigo_seguimiento`),
  KEY `idx_encomienda_remitente` (`remitente_id`,`estado`),
  KEY `idx_encomienda_destinatario` (`destinatario_id`),
  KEY `idx_encomienda_estado` (`estado`),
  KEY `idx_encomienda_anulado` (`anulado`),
  KEY `fk_enc_origen` (`agencia_origen_id`),
  KEY `fk_enc_destino` (`agencia_destino_id`),
  KEY `fk_enc_registrador` (`registrado_por_usuario_id`),
  KEY `fk_enc_emp_recep` (`empleado_recepcion_id`),
  KEY `fk_enc_emp_entrega` (`empleado_entrega_id`),
  CONSTRAINT `fk_enc_remitente` FOREIGN KEY (`remitente_id`) REFERENCES `cliente` (`id`),
  CONSTRAINT `fk_enc_destinatario` FOREIGN KEY (`destinatario_id`) REFERENCES `cliente` (`id`),
  CONSTRAINT `fk_enc_origen` FOREIGN KEY (`agencia_origen_id`) REFERENCES `agencia` (`id`),
  CONSTRAINT `fk_enc_destino` FOREIGN KEY (`agencia_destino_id`) REFERENCES `agencia` (`id`),
  CONSTRAINT `fk_enc_registrador` FOREIGN KEY (`registrado_por_usuario_id`) REFERENCES `usuario` (`id`),
  CONSTRAINT `fk_enc_emp_recep` FOREIGN KEY (`empleado_recepcion_id`) REFERENCES `usuario` (`id`),
  CONSTRAINT `fk_enc_emp_entrega` FOREIGN KEY (`empleado_entrega_id`) REFERENCES `usuario` (`id`),
  CONSTRAINT `chk_enc_recepcion` CHECK (
    (`estado` = 'PRE_REGISTRADO' AND `fecha_recepcion` IS NULL AND `costo_total` IS NULL)
    OR
    (`estado` <> 'PRE_REGISTRADO')
  ),
  CONSTRAINT `chk_enc_anulado` CHECK (
    (`anulado` = 0 AND `fecha_anulacion` IS NULL)
    OR
    (`anulado` = 1 AND `fecha_anulacion` IS NOT NULL)
  )
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

-- --------------------------------------------------------
-- Tabla `bulto`

CREATE TABLE `bulto` (
  `id` bigint(20) NOT NULL,
  `encomienda_id` bigint(20) NOT NULL,

  `tipo_paquete_id` bigint(20) NOT NULL,

  `peso_estimado` decimal(8,2) DEFAULT NULL,
  `peso_final` decimal(8,2) DEFAULT NULL,
  `costo` decimal(10,2) DEFAULT NULL,
  PRIMARY KEY (`id`),
  UNIQUE KEY `uk_bulto_numero` (`encomienda_id`,`numero_bulto`),
  KEY `fk_bulto_tipo` (`tipo_paquete_id`),
  CONSTRAINT `fk_bulto_encomienda` FOREIGN KEY (`encomienda_id`) REFERENCES `encomienda` (`id`),
  CONSTRAINT `fk_bulto_tipo` FOREIGN KEY (`tipo_paquete_id`) REFERENCES `tipo_paquete` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;

  `fecha_hora` datetime NOT NULL DEFAULT current_timestamp(),

  `usuario_id` bigint(20) DEFAULT NULL,

  PRIMARY KEY (`id`),
  KEY `idx_historial_encomienda` (`encomienda_id`,`fecha_hora`),
  KEY `fk_hist_agencia` (`agencia_id`),
  KEY `fk_hist_usuario` (`usuario_id`),
  CONSTRAINT `fk_hist_encomienda` FOREIGN KEY (`encomienda_id`) REFERENCES `encomienda` (`id`),
  CONSTRAINT `fk_hist_agencia` FOREIGN KEY (`agencia_id`) REFERENCES `agencia` (`id`),
  CONSTRAINT `fk_hist_usuario` FOREIGN KEY (`usuario_id`) REFERENCES `usuario` (`id`)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci;
